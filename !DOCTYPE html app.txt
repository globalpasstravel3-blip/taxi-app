let currentFilter="ALL";
let fullData=[];

/* LOGIN */
function login(){
  google.script.run.withSuccessHandler(res=>{
    if(res){
      loginBox.style.display="none";
      app.style.display="flex";
      load();
    }else alert("Wrong Login");
  }).checkLogin(user.value,pass.value);
}

/* NAV */
function show(x){
  dash.style.display="none";
  book.style.display="none";
  document.getElementById(x).style.display="block";
}

/* TAB */
function setTab(type){
  currentFilter=type;
  show('dash');
  applyFilter();
}

/* PAYMENT TYPE */
function handlePaymentType(){
  advance.style.display = paymentType.value==="Advance" ? "block":"none";
}

/* SAVE */
function save(){

  const data = {
    name: document.getElementById("name").value.trim(),
    phone: document.getElementById("phone").value.trim(),
    pickup: document.getElementById("pickup").value.trim(),
    drop: document.getElementById("drop").value.trim(),
    time: document.getElementById("time").value,
    car: document.getElementById("car").value,
    paymentType: document.getElementById("paymentType").value,
    fare: document.getElementById("fare").value,
    advance: document.getElementById("advance").value || 0
  };

  if(!data.name){
    alert("Name required!");
    return;
  }

  google.script.run.withSuccessHandler(()=>{
    alert("Saved");
    load();
    show('dash');
  }).saveBooking(data);
}

/* LOAD WITH REFRESH STATE */
function load(){

  const btn = document.getElementById("refreshBtn");

  if(btn){
    btn.innerText = "🔄 Refreshing...";
    btn.disabled = true;
  }

  google.script.run.withSuccessHandler(list=>{
    fullData = list;
    applyFilter();

    if(btn){
      btn.innerText = "🔄 Refresh";
      btn.disabled = false;
    }

  }).getBookings();
}

/* FILTER + DASHBOARD */
function applyFilter(){

  let html="";

  let totalProfit = 0;
  let totalDebtors = 0;
  let totalActive = 0;
  let totalCancelled = 0;

  let filtered=fullData.filter(i=>{
    if(currentFilter==="ALL") return true;
    if(currentFilter==="ACTIVE") return i.status!=="Closed" && i.status!=="Cancelled";
    if(currentFilter==="CLOSED") return i.status==="Closed";
    if(currentFilter==="CANCELLED") return i.status==="Cancelled";
    if(currentFilter==="DEBTORS") return Number(i.balance) > 0;
  });

  filtered.forEach(i=>{

    totalProfit += Number(i.profit) || 0;
    totalDebtors += Number(i.balance) || 0;

    if(i.status !== "Closed" && i.status !== "Cancelled"){
      totalActive++;
    }

    if(i.status === "Cancelled"){
      totalCancelled++;
    }

    let assignBtn="",cancelBtn="",payBtn="",closeBtn="";

    const isClosed = (i.status === "Closed");
    const isCancelled = (i.status === "Cancelled");

    let rowClass="active-row";
    if(isClosed) rowClass="closed-row";
    if(isCancelled) rowClass="cancel-row";

    if(isCancelled){
      assignBtn=`<button disabled>Assign</button>`;
      cancelBtn=`<button disabled style="background:red;color:white;">Cancelled</button>`;
      closeBtn=`<button disabled>Completed</button>`;
    }
    else if(isClosed){
      assignBtn=`<button disabled>Assign</button>`;
      cancelBtn=`<button disabled>Cancel</button>`;
      closeBtn=`<button disabled style="background:#28c76f;color:white;">Completed</button>`;
    }
    else{
      assignBtn=`<button onclick="assign(${i.row})">Assign</button>`;
      cancelBtn=`<button onclick="cancelBookingUI(${i.row})" style="background:red;color:white;">Cancel</button>`;
      closeBtn=`<button onclick="closeBooking(${i.row})">Close</button>`;
    }

    if(Number(i.balance) > 0){
      payBtn = `<button onclick="markPaid(${i.row})" style="background:#28c76f;color:white;">Receive</button>`;
    } else {
      payBtn = `<button disabled style="background:#28c76f;color:white;">Paid</button>`;
    }

    html+=`
    <tr class="${rowClass}">
      <td>${i.id}</td>
      <td>${i.name}</td>
      <td>${i.pickup}</td>
      <td>${i.status}</td>
      <td>${i.driverName||""}</td>
      <td>${i.driverPhone||""}</td>
      <td>${i.vehicleNo||""}</td>
      <td>${i.profit}</td>
      <td>${i.balance}</td>
      <td>
        <div class="action-col">
          ${assignBtn}
          ${cancelBtn}
          ${payBtn}
          ${closeBtn}
        </div>
      </td>
    </tr>`;
  });

  table.innerHTML=html;

  document.getElementById("totalProfit").innerText = totalProfit;
  document.getElementById("totalDebtors").innerText = totalDebtors;
  document.getElementById("totalActive").innerText = totalActive;
  document.getElementById("totalCancelled").innerText = totalCancelled;
}

/* ACTIONS */
function markPaid(row){
  let amount=prompt("Enter amount (leave blank for full):");
  if(amount===null) return;

  if(amount===""){
    google.script.run.withSuccessHandler(load).markFullPayment(row);
  }else{
    google.script.run.withSuccessHandler(load)
      .addPartialPayment({row:row,amount:Number(amount)});
  }
}

function assign(row){
  let d=prompt("Driver");
  let p=prompt("Phone");
  let v=prompt("Vehicle");
  let c=prompt("Cost");

  google.script.run.withSuccessHandler(load)
  .assignDriver({row,driver:d,phone:p,vehicle:v,cost:c});
}

function closeBooking(row){
  if(confirm("Close booking?")){
    google.script.run.withSuccessHandler(load).closeBooking(row);
  }
}

function cancelBookingUI(row){

  let refund = prompt("Refund Type? FULL / HALF / NONE");
  let driverPaid = prompt("Driver Paid? YES / NO");

  let driverCostPaid = 0;

  if(driverPaid === "YES"){
    driverCostPaid = prompt("Enter driver paid amount:");
  }

  google.script.run.withSuccessHandler(load).cancelBooking({
    row: row,
    refund: refund,
    driverPaid: driverPaid,
    driverCostPaid: Number(driverCostPaid)
  });
}
